\section{Why This Matters}

\begin{frame}{Why This Chapter Matters}
    \begin{columns}[c]
    \begin{column}{.47\textwidth}
        {\normalsize Your gate circuits did only what their \alert{wiring} told them. To change the behaviour, you had to rewire. A microcontroller is a digital circuit too, but a \alert{program stored in memory} controls it. The same silicon can blink an LED, measure a temperature, or drive a motor. It depends only on the bits you load.}
    \end{column}
    \begin{column}{.51\textwidth}
        \begin{block}{After this chapter, you can:}
        {\small
        \begin{enumerate}\itemsep2pt
            \item Define an embedded system and its design constraints.
            \item Distinguish a \alert{microprocessor} from a \alert{microcontroller}.
            \item Name the CPU functional units and the role of each.
            \item Explain the address, data and control buses, and compute addressable memory.
            \item Trace a short program through fetch--decode--execute.
            \item Describe how one C statement changes a voltage on a pin.
        \end{enumerate}}
        \end{block}
    \end{column}
    \end{columns}
\end{frame}

\begin{frame}{The One Question}
    \leavevmode
    This chapter introduces many new terms. Every section answers just
    one question. This question is the \alert{thread} that connects
    everything that follows:

    \vspace{4mm}
    \begin{center}
    \fbox{\parbox{.78\textwidth}{\centering\vspace{2mm}
    \large How does a program cause hardware to do something?
    \vspace{2mm}}}
    \end{center}

    \vspace{4mm}
    \begin{block}{The one fact to carry out of this chapter}
        \alert{The CPU has no special instruction for ``turn on an LED''.}

        \vspace{2mm}
        It only has instructions that \alert{read and write memory
        addresses}. Some of these addresses connect not to memory,
        but to hardware instead.

        \vspace{2mm}
        So turning on an LED means \alert{writing a number to an
        address}.
    \end{block}
\end{frame}

\begin{frame}{The Destination}
	\leavevmode

\section{Embedded Systems}
%===============================================================================

\begin{frame}{Components of the Computer}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item Since 1946 all computers have had 4 components:
            \end{itemize}
            \begin{tabular}{r L{37mm}}
            \\
                \alert{Processor} & controls the system \\ \\
                \alert{Memory} & contains program and data \\ \\
                \alert{Input} & devices that get data from the outside world into the computer \\ \\
                \alert{Output} & devices that present computation results to the outside world \\ \\
%                \alert{Bus} & links them all \\
            \end{tabular}

            \vspace{2mm}
            {\small Together they perform four functions: data \alert{processing}, \alert{storage}, \alert{movement}, and \alert{control}.}
        \end{column}
        \begin{column}{0.5\textwidth}
             \includegraphics[width=\textwidth]{assets/basic-computer.pdf}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{What Is the CPU?}
    \begin{itemize}\itemsep6pt
        \item The CPU is the \alert{brain} of the computer. It works by
              executing a \alert{program}.
        \item \textbf{Program}: a sequence of instructions that performs
              a task.
            \begin{itemize}
                \item Think of running a program like \alert{playing music}.
            \end{itemize}
        \item \textbf{Instruction}: a binary code for the simplest
              operation the processor can perform.
            \begin{itemize}
                \item Think of each instruction as \alert{one note}.
            \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Memory}
    \begin{itemize}\itemsep6pt
        \item Memory is a $k \times m$ array of stored bits
              ($k$ is usually $2^n$).
            \begin{description}
                \item[Address] unique $n$-bit identifier of a location
                \item[Contents] $m$-bit value stored in that location
            \end{description}
        \item Two basic operations:
            \begin{description}
                \item[Read] get the value from a memory location
                \item[Write] put a value into a memory location
            \end{description}
    \end{itemize}
\end{frame}

\begin{frame}{Memory: ROM and RAM}
    \begin{columns}[T]
    \begin{column}{.48\textwidth}
        \textbf{ROM --- non-volatile}
        \begin{itemize}\itemsep1pt
            \item Contents survive with \alert{no power}
            \item Holds the program: fixed at manufacture, or
                  ``burned'' with a programmer
            \item \textbf{Flash} is the dominant type: NAND or NOR
            \item STM32 devices use \alert{NOR flash} for code
        \end{itemize}
    \end{column}
    \begin{column}{.48\textwidth}
        \textbf{RAM --- volatile}
        \begin{itemize}\itemsep1pt
            \item Contents \alert{lost} without power
            \item Holds temporary data: variables, the stack
            \item \textbf{SRAM}: fast, every embedded system has it
            \item \textbf{DRAM}: cheaper, needs refresh; only larger
                  systems bother
        \end{itemize}
    \end{column}
    \end{columns}

    \vspace{3mm}
    \begin{block}{On the Nucleo-F446RE}
        \textbf{512\,KB Flash} holds your program. \textbf{128\,KB
        SRAM} holds your variables and stack. Chapter~3 returns to
        this split in detail.
    \end{block}
\end{frame}


\begin{frame}{What Makes a System Embedded}
    \leavevmode
    {\small Every computer, from server to thermostat, is processor,
    memory, and I/O. Embedded systems outnumber every other kind
    \alert{combined}: dozens per person, hidden inside everyday
    products. The difference is \alert{what the
    processor is for}. A general-purpose computer is built to run
    whatever the user installs next. An embedded system is built into
    a product to perform one job.}

    \vspace{2mm}
    \begin{columns}[T]
    \begin{column}{.48\textwidth}
        \textbf{General-purpose computer}
        \begin{itemize}
            \item Runs many different applications
            \item End-user installs new software freely
            \item \alert{Faster is always better}
            \item Sold on speed, compatibility, price
        \end{itemize}
    \end{column}
    \begin{column}{.48\textwidth}
        \textbf{Embedded system}
        \begin{itemize}
            \item Runs a few functions, fixed at design time
            \item Firmware is written once, by the designer
            \item Must meet a \alert{fixed deadline}; extra speed may not help
            \item Sold on power, cost, size, \alert{predictability}
        \end{itemize}
    \end{column}
    \end{columns}

    \vspace{3mm}
    \begin{block}{The idea that matters}
        A desktop that answers in 100\,ms instead of 10\,ms is just
        slow. But an airbag controller that fires in 100\,ms instead
        of 10\,ms has \alert{failed}.

        \vspace{1mm}
        Embedded engineering cares about \emph{predictable} time, not
        maximum speed.
    \end{block}
\end{frame}

    \begin{minipage}[c]{0.66\textwidth}

\section{Microprocessor vs Microcontroller}
%===============================================================================

\begin{frame}{SWaP-C}{The constraints that bind}
	\begin{center}
		{\small
			\striped
			\begin{tabular}{@{}l l l@{}}
				\toprule
				\textbf{Constraint} & \textbf{Why it binds} & \textbf{What it forces} \\
				\midrule
				\textbf{S}ize    & Fits inside the product it controls & Package, pin count, board area \\
				\textbf{W}eight  & Portable, airborne, wearable        & Battery size, materials \\
				\textbf{a}nd     &                                     & \\
				\textbf{P}ower   & Battery life, thermal budget        & Clock speed, sleep modes, duty cycle \\
				\textbf{C}ost    & Multiplied by production volume     & Part choice, external components \\
				\bottomrule
		\end{tabular}}
	\end{center}

	\vspace{3mm}
	\begin{block}{Cost is multiplied, not added}
		Saving RM0.50 on one part looks trivial. But at 500{,}000
		units, that saving becomes \alert{RM250{,}000}. This is why
		embedded engineers argue about single resistors.
	\end{block}

	\vspace{1mm}
	{\footnotesize \alert{Check yourself:} why might a washing machine use a microcontroller rather than a general-purpose PC?}
\end{frame}

\begin{frame}{Microprocessor and Microcontroller}
	\leavevmode
	{\small SWaP-C is a design constraint. Integrating the processor,
		memory, and peripherals onto \alert{one chip} is how that
		constraint gets met in practice: fewer parts, smaller board,
		lower cost per unit. This single-chip design is called a
		\textbf{microcontroller}.}

	\vspace{2mm}
	\begin{center}
		{\small
			\striped
			\begin{tabular}{@{}l l l@{}}
				\toprule
				\textbf{Feature} & \textbf{Microprocessor ($\mu$P)} & \textbf{Microcontroller ($\mu$C)} \\
				\midrule
				Integration  & Processor chip only & Processor, RAM, ROM, I/O on one die \\
				Flexibility  & Memory and peripherals scale & Resources fixed inside the chip \\
				Complexity   & Many components, complex PCB & Few components, simple board \\
				Power, cost  & Higher & Optimised for low power and unit cost \\
				Performance  & High clock, heavy computation & Lower clock, control and I/O \\
				\bottomrule
		\end{tabular}}
	\end{center}

	\vspace{3mm}
	\begin{block}{A microprocessor needs help; a microcontroller comes ready}
		A microprocessor generally has no RAM, no ROM, and no way to
		reach the outside world on-chip; those parts must be wired
		around it before it can do anything. A microcontroller puts
		all of this together on one chip. \alert{The Nucleo's chip is
			a microcontroller}: everything on the earlier board slide
		sits inside one package.
	\end{block}

\end{frame}

\begin{frame}{The Address Bus}
    \begin{itemize}\itemsep4pt
        \item The address bus carries the address of the memory location
              or I/O device selected for a transfer.
        \item Its \alert{width} sets the addressing range: an $n$-bit
              address bus reaches $2^n$ locations.
    \end{itemize}

    \vspace{3mm}
    \begin{center}
    {\small
    \striped
    \begin{tabular}{@{}l c r@{}}
        \toprule
        \textbf{CPU} & \textbf{Address bus width} & \textbf{Addressable memory} \\
        \midrule
        8051   & 16 bits & $2^{16}$ = 65,536 bytes = 64\,KB \\
        8086   & 20 bits & $2^{20}$ = 1,048,576 bytes = 1\,MB \\
        68000  & 24 bits & $2^{24}$ = 16,777,216 bytes = 16\,MB \\
        ARM Cortex-M4 & 32 bits & $2^{32}$ = 4,294,967,296 bytes = \alert{4\,GB} \\
        \bottomrule
    \end{tabular}}
    \end{center}
\end{frame}

\begin{frame}{The Data Bus}
    \begin{itemize}\itemsep4pt
        \item The width of the data bus sets how much data moves in
              \alert{one transfer}.
        \item Classic 8-bit microcontrollers have an 8-bit data bus.
            \begin{itemize}
                \item It moves 1 byte at a time.
                \item A 32-bit word therefore takes 4 transfers.
            \end{itemize}
        \item The ARM Cortex-M4 has a 32-bit data bus.
            \begin{itemize}
                \item It moves 4 bytes at once, so a 32-bit word takes 1 transfer.
            \end{itemize}
        \item Some chips offer an external bus of selectable width
              (8, 16 or 32 bits).
            \begin{itemize}
                \item A narrower bus is slower, but lets you use cheaper memory devices.
            \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Using the Buses: Fetching an Instruction}
    \leavevmode
    Suppose the CPU needs the instruction stored at \texttt{0x1000}:

    \vspace{2mm}
    \begin{center}
    {\small
    \begin{tabular}{@{}l l@{}}
        \toprule
        \textbf{Bus} & \textbf{Carries, this step} \\
        \midrule
        Address bus & \texttt{0x1000} --- CPU asks for this location \\
        Control bus & \texttt{READ} --- CPU asserts a read \\
        Memory      & responds with \texttt{LOAD 0x2000} \\
        Data bus    & carries that instruction back to the CPU \\
        \bottomrule
    \end{tabular}}
    \end{center}
\begin{center} \begin{tikzpicture}[ >=Stealth, block/.style={draw, rounded corners, minimum width=2.8cm, minimum height=1.6cm} ] \node[block, fill=blue!10] (cpu) {CPU}; \node[block, fill=orange!10, right=5cm of cpu] (ram) {RAM}; \draw[->, very thick, red] ([yshift=14pt]cpu.east) -- node[above] {\small Address: 0x1000} ([yshift=14pt]ram.west); \draw[<->, very thick, blue] (cpu.east) -- node[above] {\small Data: "LOAD 0x2000" instruction} (ram.west); \draw[<->, very thick, green!60!black] ([yshift=-14pt]cpu.east) -- node[above] {\small Control: READ} ([yshift=-14pt]ram.west); \end{tikzpicture} \end{center}
    \vspace{1mm}
    \begin{block}{The same pattern repeats for every data access}
        The address bus names the location. The control bus says read
        or write. The data bus carries the value. \alert{Three buses,
        one transaction.} We will use this exact pattern again when we
        trace a program.
    \end{block}
\end{frame}


\begin{frame}{Memory Organization: Addresses and Contents}
    \leavevmode
    {\small Memory is a list of numbered cells. Each cell has an
    \alert{address} and holds a value.}

    \vspace{3mm}
    \begin{columns}[c]
    \begin{column}{0.45\textwidth}
        \centering
        {\small
        \begin{tabular}{@{}c c@{}}
            \toprule
            \textbf{Address} & \textbf{Contents} \\
            \midrule
            \texttt{0x0000} & \texttt{3A} \\
            \texttt{0x0001} & \texttt{F2} \\
            \texttt{0x0002} & \texttt{17} \\
            \texttt{0x0003} & \texttt{90} \\
            \texttt{\ldots} & \texttt{\ldots} \\
            \bottomrule
        \end{tabular}}
    \end{column}
    \begin{column}{0.50\textwidth}
        {\small
        \begin{itemize}\itemsep5pt
            \item The \alert{address bus} selects one row.
            \item The \alert{data bus} carries the contents of that row.
            \item With $n$ address bits, memory has $2^n$ rows; each
                  row is $m$ bits wide.
        \end{itemize}}
    \end{column}
    \end{columns}

    \vspace{3mm}
    \begin{block}{The addresses are the same, whether the row holds RAM, Flash or a peripheral register}
        {\small That is the single fact that makes memory-mapped I/O work.}
    \end{block}
\end{frame}

    \end{block}

\begin{frame}{The CPU Faces Memory: PC, IR, AR, DR}
    \leavevmode
    {\small Four registers sit between the CPU and memory. Each one holds something that is \alert{going to} or \alert{coming from} memory.}

    \vspace{2mm}
    \begin{columns}[c]
    \begin{column}{.56\textwidth}
        \centering
        \includegraphics[width=\linewidth, height=.62\textheight, keepaspectratio]%
            {assets/cpu_inside.pdf}
    \end{column}
    \begin{column}{.42\textwidth}
        {\small
        \begin{description}\itemsep5pt
            \item[PC] Program Counter: the address of the \alert{next} instruction.
            \item[IR] Instruction Register: the instruction just fetched.
            \item[AR] Address Register: the address of the data being accessed.
            \item[DR] Data Register: the value in transit to or from that address.
        \end{description}}
    \end{column}
    \end{columns}

    \vspace{2mm}
    {\small Only these registers talk to memory. Everything else in the CPU works on what they hold.}
\end{frame}

\begin{frame}{The CPU Computes: Decoder, CU, ALU, ACC}
    \leavevmode
    {\small The rest of the CPU does the work on what the registers hold.}

    \vspace{2mm}
    \begin{columns}[c]
    \begin{column}{.56\textwidth}
        \centering
        \includegraphics[width=\linewidth, height=.62\textheight, keepaspectratio]%
            {assets/cpu_inside.pdf}
    \end{column}
    \begin{column}{.42\textwidth}
        {\footnotesize
        \begin{description}\itemsep4pt
            \item[Decoder] Reads the bit pattern in IR and works out \alert{what} operation and \alert{which} operands it names.
            \item[CU] Control Unit: generates the \alert{control signals}: which register reads, which writes, when the ALU runs.
            \item[ALU] Arithmetic Logic Unit: add, subtract, AND, OR, XOR.
            \item[ACC] Accumulator: operand and result of the ALU.
        \end{description}}
    \end{column}
    \end{columns}

    \vspace{2mm}
    {\small The ALU's bit width is ``the size of the processor'': 4 bits for a remote control, 8 for classic microcontrollers (8051, PIC), \alert{32 for the Cortex-M4}, 64 for servers.}
\end{frame}

\begin{frame}{The CPU Has an Instruction. Now What?}
    \leavevmode
    \begin{center}
    \fbox{\parbox{.8\textwidth}{\centering\vspace{2mm}
    The CPU has fetched an instruction. What does it actually do with it?
    \vspace{2mm}}}
    \end{center}

    \vspace{4mm}
    \begin{center}
    \begin{tikzpicture}[x=10mm, y=10mm]
        \node[cpublk, minimum width=26mm, minimum height=12mm] (f) at (0,0)   {Fetch};
        \node[blk,    minimum width=26mm, minimum height=12mm] (d) at (4.2,0) {Decode};
        \node[memblk, minimum width=26mm, minimum height=12mm] (e) at (8.4,0) {Execute};
        \draw[flow] (f) -- (d);
        \draw[flow] (d) -- (e);
        \draw[flow] (e.south) -- (8.4,-1.4) -- (0,-1.4) -- (f.south);
        \node[lbl, anchor=north] at (4.2,-1.45) {repeat, forever};
        \node[lbl, anchor=north, align=center] at (0,-0.75)   {What instruction?\\read from memory};
        \node[lbl, anchor=north, align=center] at (4.2,-0.75) {What does it mean?\\interpret the bit pattern};
        \node[lbl, anchor=north, align=center] at (8.4,-0.75) {What must happen?\\perform the action};
    \end{tikzpicture}
    \end{center}

    \vspace{2mm}
    {\small This is a finite state machine, driven by the clock. You
    built FSMs in digital electronics. \alert{A CPU is the largest one
    you have met so far.} The loop never stops while the machine has
    power. This is why an embedded program's \texttt{main()} is an
    infinite loop \emph{by design}.}
\end{frame}

\begin{frame}{Machine Code and Assembly Language}
    \leavevmode
    {\small Step 1 of fetch--decode--execute reads a ``machine code
    instruction.'' What is that, really?}

    \vspace{2mm}
    \begin{columns}[T]
    \begin{column}{.48\textwidth}
        \begin{block}{Machine language}
            {\small The lowest level. A bit pattern, such as
            \texttt{0001 1011}, that the Decoder is built to
            interpret. This is the \alert{only} language the CPU
            itself understands.}
        \end{block}
    \end{column}
    \begin{column}{.48\textwidth}
        \begin{block}{Assembly language}
            {\small A human-readable \alert{mnemonic} per
            instruction --- \texttt{MOV}, \texttt{ADD}, \texttt{MUL}
            --- translated to machine code by an \textbf{assembler}.
            One line in, one instruction out.}
        \end{block}
    \end{column}
    \end{columns}

    \vspace{3mm}
    {\small A \textbf{cross assembler} runs on one computer (your PC)
    and generates machine code for a different one (the Cortex-M4)
    --- the normal situation in embedded development.}

    \vspace{2mm}
    {\small Chapter 5 returns to assembly language in depth, once the
    Cortex-M4's own registers and instructions are in place.}
\end{frame}

    the compiler turns the same C line into this:}

    \vspace{2mm}
    \begin{center}
    \fbox{\parbox{.5\textwidth}{\centering\vspace{1mm}
        \textbf{C:}\quad \texttt{z = x + y;}
        \vspace{1mm}}}
    \end{center}

    \vspace{0mm}
    \begin{center}
    $\downarrow$ {\scriptsize the \alert{compiler} turns C into assembly; the \alert{assembler} turns assembly into machine code}
    \end{center}

    \vspace{0mm}
    \begin{center}
    {\small
    \striped
    \begin{tabular}{@{}l l l l@{}}
        \toprule
        \textbf{Teaching} & \textbf{Assembly} & \textbf{Machine code} & \textbf{Binary} \\

    \alert{registers}, so \texttt{y} needs its own load: one C line, four
    16-bit machine instructions. Chapters 5--7 explain each one.}
\end{frame}

\begin{frame}{We Can Now Answer the Question We Started With}
	\leavevmode
	\begin{center}
		\fbox{\parbox{.7\textwidth}{\centering\vspace{2mm}
				\large How does a program cause hardware to do something?
				\vspace{2mm}}}
	\end{center}

	\vspace{3mm}
	{\small We just traced \texttt{z = x + y;} through LOAD, ADD, and
		STORE --- ordinary memory addresses, nothing more. But the
		\alert{same} load/store mechanism works on any address. Give it
		the address of a peripheral register instead of a memory cell,
		and it changes hardware, not a variable.}
\end{frame}

\begin{frame}[fragile]{One Line of C, Back on the Board}
	\begin{columns}[c]

			Look at this line. It appears in almost every STM32 program ever
			written:

			\vspace{2mm}

				left shift.}

				statement \alert{turns on an LED}: the LED on \texttt{PA5},
