\section{Why This Matters}
%===============================================================================

\begin{frame}{Why This Chapter Matters}
    \begin{columns}[T]
    \begin{column}{.47\textwidth}
        {\normalsize Your C code reaches a hardware register, and that code compiles into instructions. Sooner or later you must read them: in a fault handler, in unexpected compiler output, in an exact timing loop, or in startup code. The next three chapters make the machine visible. The goal is not fluency, but the ability to see what the processor does under any line of C.}
    \end{column}
    \begin{column}{.51\textwidth}
        \begin{block}{After this chapter, you can:}
        {\small
        \begin{enumerate}\itemsep2pt
            \item Read an assembly program: directives, labels, instructions.
            \item Apply load--process--store to any memory variable.
            \item Choose the right load or store for byte, halfword and word; explain sign and zero extension.
            \item Trace a short sequence register by register.
            \item Explain low versus high registers and code size.
            \item Recover the C statement from a short disassembly.
        \end{enumerate}}
        \end{block}
    \end{column}
    \end{columns}
\end{frame}

\section{A First Assembly Program}
%===============================================================================

\begin{frame}[fragile]{A Complete ARM Assembly Program}
    \leavevmode
    {\small Something smaller than the blinking LED: add two constants,
    then stop. The \alert{shape} of an assembly file --- directives,
    labels, instructions --- is visible before any register touches
    hardware.}

    \vspace{2mm}
    \begin{center}
    \begin{lstlisting}[style=armasm, basicstyle=\footnotesize\ttfamily]
; Define register aliases and constants
VAL1    RN  R1               ; VAL1 is an alias for R1
VAL2    RN  R2               ; VAL2 is an alias for R2
SUM     RN  R3               ; SUM  is an alias for R3
MY_CONST EQU 0x25            ; EQU names a constant

        AREA MY_ASM_PROG, CODE, READONLY
        ENTRY                ; program entry point
__main
        MOV  VAL1, #MY_CONST ; R1 = 0x25
        MOV  VAL2, #0x34     ; R2 = 0x34
        ADD  SUM, VAL1, VAL2 ; R3 = R1 + R2

HERE    B    HERE            ; infinite loop to halt
        END                  ; end of assembly file
    \end{lstlisting}
    \end{center}
\end{frame}

    \end{tikzpicture}}
    \end{center}

    \vspace{2mm}
    {\small
    \begin{itemize}\itemsep1pt
        \item \textbf{Instructions} (\texttt{MOV}, \texttt{ADD}, \texttt{B}) become machine code.
        \item \textbf{Directives} (\texttt{RN}, \texttt{EQU}, \texttt{AREA}, \texttt{ENTRY}, \texttt{END}) are commands to the \alert{assembler}. They produce no instructions.
        \item \textbf{Labels} (\texttt{\_\_main}, \texttt{HERE}) name an address.
    \end{itemize}}

    \vspace{1mm}
    {\small \texttt{B HERE} at the end: a bare-metal program has no OS to
    return to, so an infinite loop is the normal, correct way to stop.}
\end{frame}

\begin{frame}{Tracing the Three Instructions}
    \begin{center}
    {\small
    \striped
    \begin{tabular}{@{}l l l l@{}}
        \toprule
        \textbf{After executing} & \textbf{R1} & \textbf{R2} & \textbf{R3} \\
        \midrule
        \texttt{MOV R1, \#0x25}       & \texttt{0x25} & ---           & ---           \\
        \texttt{MOV R2, \#0x34}       & \texttt{0x25} & \texttt{0x34} & ---           \\
        \texttt{ADD R3, R1, R2}       & \texttt{0x25} & \texttt{0x34} & \texttt{0x59} \\
        \bottomrule
    \end{tabular}}
    \end{center}

    \vspace{3mm}
    \begin{block}{Reading the shape of a statement}
        The first operand is the \alert{destination}. \texttt{ADD R3, R1, R2}
        reads as ``R3 becomes R1 plus R2.'' Right to left.
    \end{block}
\end{frame}

\section{The Load-Store Workflow}
%===============================================================================

\begin{frame}[plain]{Load, Process, Store}
    \begin{center}
    \includegraphics[width=.72\textwidth, height=.56\textheight, keepaspectratio]%
        {assets/loadstore-ops.pdf}
    \end{center}

    \vspace{2mm}
    {\small\centering Every operation on a memory variable follows the
    same three steps. The processor cannot operate on memory directly ---
    it only operates on registers.\par}
\end{frame}

\begin{frame}[fragile]{One C Statement, Four Instructions}
    \leavevmode
    {\small Take \texttt{c = a + b}, with all three variables in memory and
    \texttt{R1} already holding the base address of the block that contains
    them:}

    \vspace{2mm}
    \begin{center}
    \begin{lstlisting}[style=armasm, basicstyle=\footnotesize\ttfamily]
	LDR   R0, [R1]        ; load a  (memory -> register)
	LDR   R2, [R1, #4]    ; load b  (memory -> register)
	ADD   R0, R0, R2      ; process (register + register)
	STR   R0, [R1, #8]    ; store c (register -> memory)
    \end{lstlisting}
    \end{center}

    \vspace{2mm}
    {\small Four instructions for one assignment, and only one of them
    is arithmetic.}
\end{frame}

\begin{frame}[plain]{Where the Value Is, at Each Step}
    \begin{center}
    \begin{tikzpicture}[x=10mm, y=9mm]
        % Memory block
        \node[lbl, anchor=south, font=\sffamily\small] at (0.75,3.55) {\textbf{Memory}};
        \foreach \i/\nm/\val in {0/a/{7}, 1/b/{5}, 2/c/{?}}{
            \pgfmathsetmacro{\y}{2.8-\i*0.85}
            \draw[draw=accentline, fill=accentfill, thick] (0,\y) rectangle (1.5,\y+0.7);
            \node[font=\scriptsize\ttfamily] at (0.75,\y+0.35) {\val};
            \node[lbl, anchor=east, font=\scriptsize\ttfamily] at (-0.15,\y+0.35)
                {\nm \;\; \texttt{[R1,\#\pgfmathparse{int(\i*4)}\pgfmathresult]}};
        }
        % Registers
        \node[lbl, anchor=south, font=\sffamily\small] at (6.5,3.55) {\textbf{Registers}};
        \foreach \i/\rn/\val in {0/R0/{7}, 1/R2/{5}, 2/R0/{12}}{
            \pgfmathsetmacro{\y}{2.8-\i*0.85}
            \draw[draw=blkline, fill=blkfill, thick] (5.75,\y) rectangle (7.25,\y+0.7);
            \node[font=\scriptsize\ttfamily] at (6.5,\y+0.35) {\rn: \val};
        }
        \tikzset{movelbl/.style={lbl, fill=white, inner sep=2pt, font=\scriptsize\ttfamily}}

        \draw[flow, draw=blkline] (1.5,2.30) -- (5.75,2.30) node[movelbl, midway] {LDR};
        \draw[flow, draw=accentline] (5.75,1.45) -- (1.5,1.45) node[movelbl, midway, text=accentline] {STR};
        \draw[flow, draw=accentline] (7.35,3.15) -- ++(0.75,0) |- (7.35,1.45);

    \end{tikzpicture}
    \end{center}

    \vspace{2mm}
    \begin{block}{Where are \texttt{a}, \texttt{b}, \texttt{c} right now?}
        After the first \texttt{LDR}, \texttt{a} exists in \emph{two} places
        at once --- in memory and in R0. After the \texttt{ADD}, \texttt{c}
        exists only inside the CPU. Lose power before the \texttt{STR} and
        the result is gone.
    \end{block}
\end{frame}

\begin{frame}[fragile]{When the Operands Are Already in Registers}
    \leavevmode
    {\small The loads and stores are not the cost of the addition. They are
    the cost of the operands \emph{living in memory}. The same arithmetic
    on arguments already in registers:}

    \vspace{2mm}
    \begin{columns}[T]
    \begin{column}{.48\textwidth}
        \textbf{In C}

        \vspace{1mm}
        \begin{lstlisting}[style=myc]
int f(int a, int b) {
    return a + b;
}
        \end{lstlisting}

        \vspace{2mm}
        {\footnotesize Arguments arrive in \texttt{R0} and \texttt{R1} by
        the AAPCS convention. Return value leaves in \texttt{R0}.}
    \end{column}
    \begin{column}{.48\textwidth}
        \textbf{In assembly}

        \vspace{1mm}
        \begin{lstlisting}[style=armasm, basicstyle=\normalsize\ttfamily]
	ADD   R0, R0, R1
	BX    LR
        \end{lstlisting}

        \vspace{3mm}
        {\small Same addition. \alert{Two instructions} instead of four.}
    \end{column}
    \end{columns}

    \vspace{3mm}
    \begin{block}{Why the compiler works hard to keep values in registers}
        A loop that reads a global each pass is slower than one that reads
        it once into a local. Each memory access is an instruction the
        arithmetic itself does not need.
    \end{block}
\end{frame}

\section{Memory Transfer Instructions}
%===============================================================================

\begin{frame}{The Load and Store Family}
    \begin{center}
    {\small
    \striped
    \begin{tabular}{@{}l l R{.50\textwidth}@{}}
        \toprule
        \textbf{Instruction} & \textbf{Size} & \textbf{Purpose} \\
        \midrule
        \texttt{LDR} / \texttt{STR}   & word (32-bit)      & Transfer a full register \\
        \texttt{LDRB} / \texttt{STRB} & byte (8-bit)       & Transfer one byte \\
        \texttt{LDRH} / \texttt{STRH} & halfword (16-bit)  & Transfer two bytes \\
        \midrule
        \multicolumn{3}{l}{\emph{Less common: signed loads}} \\
        \texttt{LDRSB}                & byte (8-bit)       & Load a signed byte and sign-extend \\
        \texttt{LDRSH}                & halfword (16-bit)  & Load a signed halfword and sign-extend \\
        \bottomrule
    \end{tabular}}
    \end{center}

    \vspace{3mm}
    \begin{block}{Naming convention}
        No suffix means a 32-bit \alert{word}.
        \texttt{B} means \alert{byte}.
        \texttt{H} means \alert{halfword}.
        \texttt{S} marks a \alert{signed} load. There is no signed \emph{store} --- a store just copies the low bits.
    \end{block}
\end{frame}

\begin{frame}[fragile]{Word Transfers: \texttt{LDR} and \texttt{STR}}
    \begin{center}
    \includegraphics[width=.44\textwidth, keepaspectratio]{assets/inst-ldr.pdf}
    \hspace{3mm}
    \includegraphics[width=.44\textwidth, keepaspectratio]{assets/inst-str.pdf}
    \end{center}

    \vspace{1mm}
    {\small \textbf{Left:} \texttt{LDR R1, [R0]} with \texttt{R0 = 0x40020014}
    and memory bytes \texttt{EE FF C0 00}. Little-endian: the lowest-address
    byte becomes the \alert{least-significant} byte, giving
    \texttt{R1 = 0x00C0FFEE}.}

    \vspace{1mm}
    {\small \textbf{Right:} \texttt{STR R1, [R0]} with \texttt{R1 = 0x11223344}.
    The least-significant byte (\texttt{0x44}) goes to the lowest memory
    address. The same rule, in reverse.}
\end{frame}

\begin{frame}[fragile]{Byte and Halfword Loads: Zero Extension}
    \leavevmode
    {\small The destination register is always \alert{32 bits wide}. When
    fewer bytes are loaded, the processor must decide what to put in the
    unused upper bits. Ordinary byte and halfword loads fill them with
    \alert{zeros}.}

    \vspace{2mm}
    \begin{center}
    \includegraphics[width=.78\textwidth]{assets/bits-ldrb.pdf}

    \vspace{3mm}
    \includegraphics[width=.78\textwidth]{assets/bits-ldrh.pdf}
    \end{center}

    \vspace{1mm}
    {\small \texttt{LDRB} fills the upper 24 bits with zero;
    \texttt{LDRH} fills the upper 16 bits with zero.}
\end{frame}

\begin{frame}[fragile]{Byte and Halfword Stores}
    \leavevmode
    {\small The store side is symmetric and simpler. \texttt{STRB} writes
    only the low 8 bits of the register; \texttt{STRH} writes only the low
    16 bits. The upper bits are \alert{ignored}.}

    \vspace{3mm}
    \begin{columns}[T]
    \begin{column}{.48\textwidth}
        \begin{lstlisting}[style=armasm, basicstyle=\normalsize\ttfamily]
	MOV  R0, #0x11223344
	STRB R0, [R1]
        \end{lstlisting}

        \vspace{2mm}
        {\small Stores only \texttt{0x44}. The upper 24 bits are ignored.}
    \end{column}
    \begin{column}{.48\textwidth}
        \begin{lstlisting}[style=armasm, basicstyle=\normalsize\ttfamily]
	MOV  R0, #0x12345678
	STRH R0, [R1]
        \end{lstlisting}

        \vspace{2mm}
        {\small Stores only \texttt{0x5678}. The upper 16 bits are ignored.}
    \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Common Memory Access Instructions}
    \begin{center}
    {\small
    \striped
    \begin{tabular}{@{}l l l l@{}}
        \toprule
        \textbf{Instruction} & \textbf{Size} & \textbf{Extension} & \textbf{Operation} \\
        \midrule
        \texttt{LDR Rd, [Rn]}   & 32-bit word     & None        & $\text{Rd}=[\text{Rn}]$ \\
        \texttt{STR Rt, [Rn]}   & 32-bit word     & None        & $[\text{Rn}]=\text{Rt}$ \\
        \texttt{LDRB Rd, [Rn]}  & 8-bit byte      & Zero-extend & $\text{Rd}=\text{ZeroExtend}([\text{Rn}]_{\text{byte}})$ \\
        \texttt{STRB Rt, [Rn]}  & 8-bit byte      & None        & $[\text{Rn}]_{\text{byte}}=\text{Rt}[7{:}0]$ \\
        \texttt{LDRH Rd, [Rn]}  & 16-bit halfword & Zero-extend & $\text{Rd}=\text{ZeroExtend}([\text{Rn}]_{\text{half}})$ \\
        \texttt{STRH Rt, [Rn]}  & 16-bit halfword & None        & $[\text{Rn}]_{\text{half}}=\text{Rt}[15{:}0]$ \\
        \bottomrule
    \end{tabular}}
    \end{center}
\end{frame}

\section{Optional: Signed Loads}
%===============================================================================

\begin{frame}{Signed Loads: \texttt{LDRSB} and \texttt{LDRSH}}
    \leavevmode
    {\small Many embedded programs never need these. Peripheral registers,
    status bits, and addresses are naturally unsigned.}

    \vspace{2mm}
    {\small ARM still provides them. Instead of filling the upper bits with
    zeros, a \alert{signed} load copies the source's \alert{sign bit} into
    the upper bits. This is called \emph{sign extension}.}

    \vspace{3mm}
    \begin{center}
    \includegraphics[width=.78\textwidth]{assets/bits-ldrsb.pdf}

    \vspace{3mm}
    \includegraphics[width=.78\textwidth]{assets/bits-ldrsh.pdf}
    \end{center}
\end{frame}

\begin{frame}[fragile]{The Same Bits, Different Numbers}
    \leavevmode
    {\small Suppose memory contains the byte \texttt{0x8C}. Its binary form
    is \texttt{10001100}. Bit 7 is 1, so this byte represents the signed
    value \alert{$-116$}.}

    \vspace{3mm}
    \begin{center}
    {\small
    \striped
    \begin{tabular}{@{}l l R{.42\textwidth}@{}}
        \toprule
        \textbf{Instruction} & \textbf{R0 after} & \textbf{Meaning} \\
        \midrule
        \texttt{LDRB  R0, [R1]} & \texttt{0x0000008C} & decimal 140 \\
        \texttt{LDRSB R0, [R1]} & \texttt{0xFFFFFF8C} & decimal $-116$ \\
        \bottomrule
    \end{tabular}}
    \end{center}

    \vspace{3mm}
    {\small The \alert{same eight memory bits} can mean different numbers,
    depending on which load you use.}

    \vspace{2mm}
    {\small If the sign bit is zero, both instructions give the same
    result. Loading \texttt{0x4F} yields \texttt{0x0000004F} either way.}

    \vspace{3mm}
    \begin{block}{C connection}
        A C compiler picks \texttt{LDRB} for \texttt{uint8\_t} and
        \texttt{LDRSB} for \texttt{int8\_t}. The C type decides which
        instruction appears.
    \end{block}
\end{frame}

\section{Constants: \texttt{MOV}, \texttt{MVN}, and Building an Address}
%===============================================================================

\begin{frame}[fragile]{\texttt{MOV} and \texttt{MVN}}
    \leavevmode
    {\small Two instructions that never touch memory.}

    \vspace{2mm}
    {\small \textbf{\texttt{MOV Rd, Operand2}} copies its second operand
    into \texttt{Rd} --- an encodable constant, or another register.
    \textbf{\texttt{MVN}} does the same but inverts every bit on the way.}

    \vspace{2mm}
    \begin{center}
    \begin{lstlisting}[style=armasm, basicstyle=\normalsize\ttfamily]
	MOV R0, R1       ; R0 = R1
	MOV R0, #25      ; R0 = 25
	MVN R0, #0       ; R0 = 0xFFFFFFFF
	MVN R0, R1       ; R0 = ~R1  (one's complement)
    \end{lstlisting}
    \end{center}

    \vspace{2mm}
    \begin{block}{The encoding limit}
        \texttt{MOV} accepts only a restricted 12-bit encoding --- an 8-bit
        value plus a rotation. So \texttt{MOV R0, \#0xFF000000} is
        encodable; \texttt{MOV R0, \#0x12345678} is \alert{not}. Every
        peripheral address falls in the second category.
    \end{block}
\end{frame}

\begin{frame}[fragile]{Loading a Large Constant: \texttt{LDR Rd, =value}}
    \leavevmode
    {\small The form you write by hand. It is not a real instruction ---
    it is a \alert{pseudo-instruction}.}

    \vspace{2mm}
    \begin{center}
    \begin{lstlisting}[style=armasm, basicstyle=\large\ttfamily]
	LDR R0, =0x40020014
    \end{lstlisting}
    \end{center}

    \vspace{3mm}
    {\small You say \emph{what} value you want. The assembler decides
    \emph{how} to build it:}

    \vspace{1mm}
    {\small
    \begin{itemize}\itemsep1pt
        \item A single \texttt{MOV}, if the value happens to be encodable.
        \item Otherwise, a \alert{literal pool}: a small block of data
              placed in the code section, read with a PC-relative load.
    \end{itemize}}

    \vspace{3mm}
    \begin{block}{Programmer vs assembler}
        Write \texttt{LDR Rd, =value} and let the assembler decide. Read
        the other forms and know what they mean.
    \end{block}
\end{frame}

\begin{frame}[fragile]{Building a 32-Bit Constant with \texttt{MOVW}/\texttt{MOVT}}
    \leavevmode
    {\small One common way a Thumb-2 compiler builds a 32-bit constant:
    one halfword at a time. \texttt{MOVW} writes the low 16 bits and
    clears the high half. \texttt{MOVT} writes the high 16 bits, leaving
    the low half alone.}

    \vspace{2mm}
    \begin{center}
    \begin{lstlisting}[style=armasm, basicstyle=\footnotesize\ttfamily]
; build 0x40020014 (GPIOA->ODR), one halfword at a time
	MOVW R0, #0x0014   ; R0 = 0x00000014, high half cleared
	MOVT R0, #0x4002   ; R0 = 0x40020014, high half set, low half kept
    \end{lstlisting}
    \end{center}

    \vspace{2mm}
    \begin{center}
        \large\texttt{R0 = 0x40020014}
    \end{center}

    \vspace{2mm}
    \begin{block}{A pair to recognise, not one to write}
        {\small \texttt{MOVW}/\texttt{MOVT} is a \alert{common outcome},
        not a rule. A compiler also picks a plain \texttt{MOV} for small
        constants, a shift trick, or a literal pool --- depending on the
        value, optimization level, and target.}
    \end{block}
\end{frame}

\begin{frame}[fragile]{Reading It Back}
    \leavevmode
    \begin{center}
    \begin{lstlisting}[style=armasm, basicstyle=\normalsize\ttfamily]
	MOVW R3, #0x0014
	MOVT R3, #0x4002   ; "R3 is being given a 32-bit address"
	LDR  R2, [R3]      ; "...and now something is read through it"
    \end{lstlisting}
    \end{center}

    \vspace{3mm}
    \begin{block}{The useful reading is always the same}
        Three lines, \alert{one address being built, then used}. The exact
        halves rarely matter. Seeing it as one operation, not three
        unrelated ones, is the whole skill.
    \end{block}

    \vspace{3mm}
    {\small \textbf{C connection:} a declaration as ordinary as
    \texttt{uint32\_t x = 0x12345678;} can require two instructions.
    Nothing in the C source hints at the split. It is a property of the
    \alert{instruction encoding}.}
\end{frame}

\begin{frame}{A Look Ahead: The Barrel Shifter}
    \leavevmode
    {\small ARM can shift a register on its way \emph{into} an
    instruction, at no extra cost --- for example
    \texttt{ADD R0, R1, R2, LSL \#2}.}

    \vspace{3mm}
    {\small We return to this feature in Chapter~6, where it becomes
    particularly useful for calculating addresses. It is mentioned here
    only so that a shifted operand in later disassembly does not come as
    a surprise.}
\end{frame}

\section{Worked Example: Reading Your First Disassembly}
%===============================================================================

\begin{frame}[fragile]{Your First Disassembly}
    \begin{columns}[T]
    \begin{column}{.46\textwidth}
        \textbf{The C}

        \vspace{1mm}
        \begin{lstlisting}[style=myc]
int c;
c = a + b;
        \end{lstlisting}

        \vspace{3mm}
        \textbf{The listing}

        \vspace{1mm}
        \begin{lstlisting}[style=armasm, basicstyle=\footnotesize\ttfamily]
	LDR  R3, [R7, #4]
	LDR  R2, [R7, #8]
	ADD  R3, R3, R2
	STR  R3, [R7, #12]
        \end{lstlisting}
    \end{column}
    \begin{column}{.5\textwidth}
        \textbf{Reading it back}

        \vspace{1mm}
        {\small
        \begin{enumerate}\itemsep1pt
            \item \texttt{R7} is the \alert{frame pointer}, so these are
                  \alert{local variables} at offsets 4, 8, 12.
            \item Load two of them.
            \item Add them.
            \item Store the result in a third.
        \end{enumerate}}

        \vspace{3mm}
        \begin{block}{Three things to notice}
            {\small Same \alert{load--process--store} shape as every other
            example. No \texttt{S} suffix --- nothing branches on the
            result. Offsets from \texttt{R7} say \alert{stack locals}, not
            globals.}
        \end{block}
    \end{column}
    \end{columns}
\end{frame}

\begin{frame}{One Operation, Viewed at Five Levels}
    \begin{center}
    {\small
    \striped
    \begin{tabular}{@{}l R{.65\textwidth}@{}}
        \toprule
        \textbf{Level} & \textbf{Representation} \\
        \midrule
        Physical hardware & Two register-file read ports feeding the ALU's 32-bit adder \\
        Register          & Two locals loaded, added, one result stored \\
        C                 & \texttt{c = a + b;} \\
        Assembly          & \texttt{LDR R3, [R7, \#4]}; \texttt{LDR R2, [R7, \#8]}; \texttt{ADD R3, R3, R2}; \texttt{STR R3, [R7, \#12]} \\
        Machine code      & Four Thumb-2 encodings stored in Flash \\
        \bottomrule
    \end{tabular}}
    \end{center}

    \vspace{3mm}
    \begin{block}{You can now}
        read a short assembly program, follow a value through the
        load--process--store cycle, and choose the correct load or store
        for any data width.
    \end{block}
\end{frame}

		\end{itemize}
	\end{itemize}

		STRB R0, [R1]
	\end{verbatim}
	\centering
	{\small\itshape Before: R1 = 0x11223344, \quad [0x40020014] = 0x00C0FFEE\par}
	\vspace{2mm}
	\resizebox{0.92\textwidth}{!}{%
		\begin{tikzpicture}[x=10mm, y=8mm]

			% --- Register R1: only byte0 (low byte) is stored ---
			\foreach \p/\val/\used in {0/{0x11}/0, 1/{0x22}/0, 2/{0x33}/0, 3/{0x44}/1}{
				\pgfmathsetmacro{\x}{\regx+\p*\bw}
				\ifnum\used=1
				\draw[loadedbox] (\x,\regy) rectangle ({\x+\bw},{\regy-\rh});
				\else
				\draw[extbox] (\x,\regy) rectangle ({\x+\bw},{\regy-\rh});
				\fi
				\node[font=\ttfamily\small] at ({\x+\bw/2},{\regy-\rh/2}) {\val};
			}
			\node[lbl, anchor=east, font=\sffamily\small] at ({\regx-0.3},{\regy-\rh/2}) {R1};

			% under-brace: upper 3 bytes not stored
			\draw[decorate, decoration={brace, amplitude=4pt, mirror}, draw=blkline]
			({\regx},{\regy-\rh-0.1}) -- ({\regx+3*\bw},{\regy-\rh-0.1});
			\node[lbl, font=\sffamily\scriptsize, text=blkline, align=center]
			at ({\regx+1.5*\bw},{\regy-\rh-0.55}) {bits 31:8\\(not stored)};

			% --- Memory: only the low byte overwritten, others keep their old value ---
			\foreach \i/\val/\addr/\written in {%
				0/{0x44}/{0x40020014}/1,
				1/{0xFF}/{0x40020015}/0,
				2/{0xC0}/{0x40020016}/0,
				3/{0x00}/{0x40020017}/0}{
				\pgfmathsetmacro{\y}{\memy-\i*\rh}
				\ifnum\written=1
				\draw[memwrite] (\memx,\y) rectangle ({\memx+\bw},{\y-\rh});
				\else
				\draw[memkeep] (\memx,\y) rectangle ({\memx+\bw},{\y-\rh});
				\fi
				\node[font=\ttfamily\small] at ({\memx+\bw/2},{\y-\rh/2}) {\val};
				\node[lbl, anchor=west, font=\sffamily\scriptsize] at ({\memx+\bw+0.3},{\y-\rh/2}) {\addr};
			}
			\node[lbl, anchor=south, font=\sffamily\small] at ({\memx+\bw/2},{\memy+0.15}) {Memory};

			% under-brace: unchanged memory bytes
			\draw[decorate, decoration={brace, amplitude=4pt, mirror}, draw=accentline]
			({\memx-0.1},{\memy-1*\rh}) -- ({\memx-0.1},{\memy-4*\rh});
			\node[lbl, font=\sffamily\scriptsize, text=accentline, align=center]
			at ({\memx-\bw/2-.4},{\memy-2*\rh-0.5}) {unchanged};

			% --- Arrow: 1 byte flows down from register into memory ---
			\draw[flow, draw=blkline] ({\regx+3*\bw+\bw/2},{\regy-\rh}) |- (\memx,{\memy-0*\rh-\rh/2});

			% --- Address axis + brace ---
			\def\axisx{9.1}
			\draw[-{Stealth[length=2.2mm]}, draw=black!45]
			({\axisx},{\memy+0.05}) -- ({\axisx},{\memy-4*\rh-0.05});
			\node[lbl, anchor=south, rotate=90, font=\sffamily\scriptsize]
			at ({\axisx-0.15},{\memy-2*\rh}) {increasing address};
			\draw[decorate, decoration={brace, amplitude=4pt}, draw=accentline]
			({\axisx+0.6},{\memy}) -- ({\axisx+0.6},{\memy-1*\rh});
			\node[lbl, anchor=west, align=left, font=\sffamily\scriptsize, text=accentline]
			at ({\axisx+0.85},{\memy-0.5*\rh}) {1 byte\\written by\\\texttt{STRB}};

		\end{tikzpicture}%
	}
	\begin{itemize}
		\item After \texttt{STRB R1, [R0]} runs, the \alert{memory byte} at 0x40020014 holds 0x44.
	\end{itemize}
\end{frame}
